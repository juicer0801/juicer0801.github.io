// 1. 指针和数组的区别
int a[10];
int *p = a;
sizeof(a); // 40（假设 int 4 字节）
sizeof(p); // 8 或 4，指针大小

// 2. sizeof 和 strlen
char s[] = “hello”;
sizeof(s); // 6，包含 ‘\0’
strlen(s); // 5，不包含 ‘\0’

// 3. const 位置
const char p; // 指向的内容不可改

const char * const p; // 都不可改

// 字符串反转
void reverse(char s) {
char p = s, q = s + strlen(s) - 1;
while (p < q) {
char t = p; p++ = q; *q— = t;
}
}

// 链表反转
ListNode reverse(ListNode head) {
ListNode prev = NULL;
while (head) {
ListNode next = head->next;
head->next = prev;
prev = head;
head = next;
}
return prev;
}

// 快慢指针判断链表有环
int hasCycle(ListNode head) {
ListNode slow = head, *fast = head;
while (fast && fast->next) {
slow = slow->next;
fast = fast->next->next;
if (slow == fast) return 1;
}
return 0;
}

// 二分查找
int binary_search(int a[], int n, int target) {
int l = 0, r = n - 1;
while (l <= r) {
int mid = l + (r - l) / 2; // 防溢出
if (a[mid] == target) return mid;
if (a[mid] < target) l = mid + 1;
else r = mid - 1;
}
return -1;
}

// 实现 atoi
int my_atoi(const char s) {
int sign = 1, num = 0;
while (s == ‘ ‘) s++;
if (s == ‘-‘) { sign = -1; s++; }
else if (s == ‘+’) s++;
while (s >= ‘0’ && s <= ‘9’) {
num = num 10 + (s - ‘0’);
s++;
}
return sign * num;
}
