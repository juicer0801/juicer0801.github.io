int *p; *p=10; // 野指针，p是野指针，未指向有效内存
free(p); p=NULL; // 防悬空

int p = malloc(nsizeof(*p)); // 推荐 sizeof(*p)
free(p);
free(p); // 错：重复 free
p = NULL; // 好习惯：free 后置空

void f(int *p) { p = malloc(sizeof(int));} // 只改局部，需 int ** 或返回指针

const char *p1; // 指向的内容不可改
char * const p2 = buf; // 指针本身不可改
const char * const p3 = “x”; // 两者都不可改

char *f() {
char buf[10];
return buf; // 错：返回局部数组，函数结束就失效
}

int a[10];
void f(int a[]) {
sizeof(a); // 这里是指针大小，不是数组总大小
}
