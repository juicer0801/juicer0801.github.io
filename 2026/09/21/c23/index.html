回调函数
typedef void (*Callback)(int);

void for_each(int *a, int n, Callback cb) {
for (int i = 0; i < n; i++) cb(a[i]);
}

void print(int x) { printf(“%d “, x); }
for_each(arr, n, print);

状态机
typedef enum { IDLE, RUN, STOP } State;
typedef enum { START, HALT } Event;

State state = IDLE;

void fsm(Event e) {
switch (state) {
case IDLE:
if (e == START) state = RUN;
break;
case RUN:
if (e == HALT) state = STOP;
break;
case STOP:
if (e == START) state = RUN;
break;
}
}

面向对象
typedef struct Shape Shape;

struct Shape {
void (*draw)(Shape *);
void (*area)(Shape *);
};

typedef struct {
Shape base;
int r;
} Circle;

void circle_draw(Shape *s) { printf(“circle\n”); }
void circle_area(Shape *s) {
Circle *c = (Circle *)s;
printf(“%f\n”, 3.14 * c->r * c->r);
}

单例
Circle c = { { circle_draw, circle_area }, 5 };
c.base.draw((Shape *)&c);

策略模式
typedef int (*Compare)(const void *, const void *);

void sort(int *a, int n, Compare cmp) {
qsort(a, n, sizeof(int), cmp);
}

int asc(const void *a, const void b) {
return ((int *)a > *(int )b) - ((int *)a < *(int *)b);
}

int desc(const void *a, const void b) {
return ((int *)b > *(int )a) - ((int *)b < *(int *)a);
}

观察者模式
typedef struct Observer {
void (*update)(struct Observer *, int);
struct Observer *next;
} Observer;

void notify(Observer *list, int val) {
for (; list; list = list->next)
list->update(list, val);
}
