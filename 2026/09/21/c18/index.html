易错点
1.
// 链表删除节点，忘了释放
while (head) {
Node *tmp = head;
head = head->next;
free(tmp); // 必须 free
}

2.
// 单链表反转，指针丢失
Node cur = head, prev = NULL, *next;
while (cur) {
next = cur->next; // 先保存
cur->next = prev;
prev = cur;
cur = next;
}
head = prev;

3.
// 动态数组扩容，忘记更新容量
if (size == cap) {
cap = 2;
arr = realloc(arr, cap sizeof(*arr));
}

小技巧
1，
// 带头节点的链表，简化插入删除
typedef struct Node {
int val;
struct Node *next;
} Node;

2.
// 动态数组（vector 风格）
typedef struct {
int *data;
size_t size;
size_t cap;
} Vec;

void vec_push(Vec v, int x) {
if (v->size == v->cap) {
v->cap = v->cap ? v->cap 2 : 4;
v->data = realloc(v->data, v->cap sizeof(v->data));
}
v->data[v->size++] = x;
}

3.
// 用数组实现栈
int stack[100], top = -1;
stack[++top] = x; // push
x = stack[top—]; // pop

4.
// 用数组实现循环队列

define N 100

int q[N], head = 0, tail = 0;
q[tail] = x; tail = (tail + 1) % N; // enqueue
x = q[head]; head = (head + 1) % N; // dequeue

5.
// 哈希表简单实现

define SIZE 1000

typedef struct HNode {
int key, val;
struct HNode next;
} HNode;
HNode table[SIZE]; // 拉链法

6.
// 二叉树遍历（递归）
void inorder(TreeNode *root) {
if (!root) return;
inorder(root->left);
printf(“%d “, root->val);
inorder(root->right);
}
