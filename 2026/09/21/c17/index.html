易错点
1.
gets(buf); // 极危险，C11好像已移除
scanf(“%s”, buf); // 不限制长度，可能溢出
scanf(“%19s”, buf); // 限制宽度，稍微i好一点

char buf[5];
strcpy(buf, “hello”); // 溢出
strncpy(buf, “hello”, sizeof(buf));
// strncpy 可能不补 ‘\0’
buf[sizeof(buf) - 1] = ‘\0’;

int x = atoi(“abc”); // 返回 0，无法区分错误和真的 0

4，
int *p = malloc(n * sizeof(int));
// 不检查 NULL
free(p);
free(p); // 重复 free

memcpy(dst, src, n); // 如果 dst 和 src 重叠，未定义
memmove(dst, src, n); // 重叠时用这个

int cmp(const void *a, const void *b) {
return (int)a - (int)b; // 可能溢出
}

fflush(stdin); // 未定义行为，不要用

小技巧
1.
int n = snprintf(buf, sizeof(buf), “%s”, str);
if (n < 0 || n >= (int)sizeof(buf)) {
// 被截断
}

2，
char *end;
errno = 0;
long v = strtol(s, &end, 10);
if (errno != 0 || *end != ‘\0’) {
// 转换失败
}

int *tmp = realloc(p, new_size);
if (!tmp) {
// 失败，p 仍有效
} else {
p = tmp;
}

int *p = calloc(n, sizeof(*p)); // 自动清零

int cmp(const void *a, const void *b) {
int x = (const int)a;
int y = (const int)b;
return (x > y) - (x < y); // 避免溢出
}
qsort(arr, n, sizeof(arr[0]), cmp);

fgets(buf, sizeof(buf), stdin);
buf[strcspn(buf, “\n”)] = ‘\0’;

// POSIX 可用 getline 动态读整行
char *line = NULL;
size_t cap = 0;
ssize_t len = getline(&line, &cap, stdin);
free(line);
