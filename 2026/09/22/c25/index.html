// 假设 long 是 32 位
// 假设指针和 int 同大小
// 直接二进制读写结构体，跨平台可能错
// 忽略对齐，导致性能下降或崩溃
// 用 char 存 EOF，可能出错
int c; while ((c = fgetc(fp)) != EOF) { }

#include <stdint.h>
int8_t a;
uint16_t b;
int32_t c;
uint64_t d;

// 不要假设 int 大小
sizeof(int); // 可能是 2、4、8
sizeof(long); // Windows 4，Linux 64 位 8
sizeof(void *); // 32 位 4，64 位 8

// 条件编译
#if defined(_WIN32)
// Windows
#elif defined(linux)
// Linux
#elif defined(APPLE)
// macOS
#endif

// 网络字节序转换
uint32_t net = htonl(host);
uint32_t host = ntohl(net);

// 结构体紧凑
#pragma pack(push, 1)
typedef struct {
uint8_t a;
uint32_t b;
} Packed;
#pragma pack(pop)

#include <assert.h>
static_assert(sizeof(int) == 4, “int must be 4 bytes”);

// 路径和换行
#ifdef _WIN32
#define PATH_SEP ‘\‘
#define NEWLINE “\r\n”
#else
#define PATH_SEP ‘/‘
#define NEWLINE “\n”
#endif

// 线程和时间
#ifdef _WIN32
#include <windows.h>
#define sleep_ms(ms) Sleep(ms)
#else
#include <unistd.h>
#define sleep_ms(ms) usleep((ms) * 1000)
#endif
